cmake_minimum_required(VERSION 3.16)
project(vita_pthread_ng C)
set(VPTHREAD_BACKEND "sce-sim" CACHE STRING "HAL backend: sce-sim, vita or vita-scelibc")
option(VPTHREAD_DIAGNOSTIC_NO_ASYNC_GUARD_ATOMIC "Use non-atomic async_guard for Vita diagnostics" OFF)
add_library(vpthread STATIC src/thread.c src/cancel.c src/mutex.c src/cond.c src/tsd.c src/rwlock.c src/barrier.c src/once.c src/posix.c)
target_include_directories(vpthread PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(vpthread PUBLIC c_std_11)
target_compile_options(vpthread PRIVATE -Wall -Wextra -Werror -Wshadow -Wconversion)
if(VPTHREAD_BACKEND STREQUAL "sce-sim")
  target_sources(vpthread PRIVATE src/sce_sim/sce_sim.c src/posix_cp_sim.c)
  target_compile_definitions(vpthread PUBLIC VPTHREAD_HAL_SCE_SIM=1)
  find_package(Threads REQUIRED)
  target_link_libraries(vpthread PUBLIC Threads::Threads)
elseif(VPTHREAD_BACKEND STREQUAL "vita")
  target_sources(vpthread PRIVATE src/posix_sem_vita.c)
  target_compile_definitions(vpthread PUBLIC VPTHREAD_HAL_VITA=1)
  include("${VITASDK}/share/vita.cmake" REQUIRED)
elseif(VPTHREAD_BACKEND STREQUAL "vita-scelibc")
  target_sources(vpthread PRIVATE src/posix_sem_vita.c)
  target_compile_definitions(vpthread PUBLIC VPTHREAD_HAL_VITA_SCELIBC=1)
  # semaphore.h/pthread.h are not part of SceLibc; ship vpthread's own.
  target_include_directories(vpthread PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include/vpthread/world)
  include("${VITASDK}/share/vita.cmake" REQUIRED)
else()
  message(FATAL_ERROR "Unknown VPTHREAD_BACKEND=${VPTHREAD_BACKEND}")
endif()
if(VPTHREAD_DIAGNOSTIC_NO_ASYNC_GUARD_ATOMIC)
  if(NOT VPTHREAD_BACKEND STREQUAL "vita")
    message(FATAL_ERROR "VPTHREAD_DIAGNOSTIC_NO_ASYNC_GUARD_ATOMIC requires VPTHREAD_BACKEND=vita")
  endif()
  target_compile_definitions(vpthread PRIVATE VPTHREAD_DIAGNOSTIC_NO_ASYNC_GUARD_ATOMIC=1)
endif()

  elseif(VPTHREAD_BACKEND STREQUAL "vita")
   function(add_vita_test test_name test_source is_posix test_index)
    math(EXPR test_suffix "10000 + ${test_index}")
    string(SUBSTRING "${test_suffix}" 1 4 test_suffix)
    set(title_id "VPTHR${test_suffix}")
    set(target "vpthread-vita-${test_name}")
    add_executable(${target} tests/vita_test_main.c ${test_source})
    set_source_files_properties(${test_source} PROPERTIES COMPILE_DEFINITIONS main=vp_vita_test_main)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_compile_definitions(${target} PRIVATE
     VP_VITA_TEST_MARKER="ux0:data/${title_id}.result"
     VP_VITA_TEST_NAME="${test_name}"
    )
    if(is_posix)
     target_include_directories(${target} BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include/vpthread/abi)
    endif()
    target_link_libraries(${target} PRIVATE vpthread SceLibKernel_stub SceKernelThreadMgr_stub)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-return-type)
    set(VITA_TITLEID ${title_id})
    set(VITA_APP_NAME "vpthread ${test_name}")
    set(VITA_VERSION "01.00")
    vita_create_self(${target}.self ${target})
    vita_create_vpk(${target}.vpk ${VITA_TITLEID} ${target}.self
     VERSION ${VITA_VERSION}
     NAME ${VITA_APP_NAME}
    )
   endfunction()
